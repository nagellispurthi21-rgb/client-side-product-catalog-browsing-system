```javascript
async function loadProducts() {
    const response = await fetch("/api/products");
    const products = await response.json();

    const container = document.getElementById("products");

    products.forEach(product => {
        container.innerHTML += `
            <div class="card">
                <h3>${product.name}</h3>
                <p>₹${product.price}</p>
            </div>
        `;
    });
}

loadProducts();
```
